# ============================================================
# Modelo de .htaccess para pagina estatica em servidor Apache.
# Vale em hospedagem compartilhada (cPanel, Hostinger e afins)
# e na Cloudways (onde o nginx serve css/js SEM passar por aqui:
# por isso o carimbo ?v=<timestamp> nos links continua obrigatorio).
# ============================================================

# --- Enquanto e rascunho. Remover no dia do lancamento. ---
# Lembre que a meta tag no HTML e um segundo lugar: tire os dois.
<IfModule mod_headers.c>
  Header set X-Robots-Tag "noindex, nofollow"
</IfModule>

# --- Forcar https ---
# ATENCAO: atras de proxy (Cloudways, qualquer nginx na frente) o Apache
# enxerga a requisicao como http mesmo quando o visitante veio por https.
# Testar %{HTTPS} ali faz LOOP de redirect. O sinal certo e o header que o
# proxy manda, X-Forwarded-Proto. Esta regra funciona nos dois cenarios.
<IfModule mod_rewrite.c>
  RewriteEngine On
  RewriteCond %{HTTP:X-Forwarded-Proto} !https
  RewriteCond %{HTTPS} !=on
  RewriteRule ^ https://%{HTTP_HOST}%{REQUEST_URI} [R=301,L]
</IfModule>

# --- Listagem de diretorio: nunca ---
Options -Indexes

# --- Headers de seguranca ---
# Ver references/auditoria-seguranca.md pra o que cada um evita.
<IfModule mod_headers.c>
  Header always set Strict-Transport-Security "max-age=31536000; includeSubDomains"
  Header always set X-Content-Type-Options "nosniff"
  Header always set X-Frame-Options "SAMEORIGIN"
  Header always set Referrer-Policy "strict-origin-when-cross-origin"
  Header always set Permissions-Policy "camera=(), microphone=(), geolocation=()"
  Header unset X-Powered-By
</IfModule>

# --- Cache por tipo de arquivo ---
# Midia e fonte podem ser imutaveis: o nome muda quando o conteudo muda.
# css e js NAO: mesmo nome, conteudo novo a cada envio.
<IfModule mod_headers.c>
  <FilesMatch "\.(jpg|jpeg|png|gif|webp|avif|svg|mp4|webm|woff2)$">
    Header set Cache-Control "public, max-age=31536000, immutable"
  </FilesMatch>

  <FilesMatch "\.(css|js)$">
    Header set Cache-Control "public, max-age=0, must-revalidate"
  </FilesMatch>

  # HTML nunca em cache: e o arquivo que aponta pra todos os outros.
  # Em servidor com proxy na frente, isso e o que evita ficar horas
  # servindo a versao velha.
  <FilesMatch "\.html$">
    Header set Cache-Control "no-cache, must-revalidate"
  </FilesMatch>
</IfModule>

# --- Compressao ---
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE text/html text/css text/plain
  AddOutputFilterByType DEFLATE application/javascript application/json
  AddOutputFilterByType DEFLATE image/svg+xml
</IfModule>

# --- Tipos que servidor antigo nao conhece ---
<IfModule mod_mime.c>
  AddType image/avif .avif
  AddType image/webp .webp
  AddType font/woff2 .woff2
</IfModule>

# --- Video precisa aceitar range, senao nao da pra pular no meio ---
<IfModule mod_headers.c>
  <FilesMatch "\.(mp4|webm)$">
    Header set Accept-Ranges bytes
  </FilesMatch>
</IfModule>
